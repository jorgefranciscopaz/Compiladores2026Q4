int total = 0;
for (int i = 1; i <= 10; i = i + 1) {
    bool even = i - (i / 2) * 2 == 0;
    if (even && i > 2) {
        total = total + i;
    }
}
print total;          // 4 + 6 + 8 + 10 = 28
